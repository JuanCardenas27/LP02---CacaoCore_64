func fib(n : int){
     let f_n1 : int = 0
     let f_n2 : int = 0

     if n == 0 or n == 1{
	deliver 1
     }
     otherwise {
	        set f_n1 = fib(n-1)
            set f_n2 = fib(n-2)
	        set f_n1 += f_n2
	deliver f_n1
     }
}

let res : int
for (let i: int = 0, i < 5, set i += 1){
    set res = fib(i)
    show res
}
